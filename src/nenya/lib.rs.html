1//! # Distributed Rate Limiting System
2//!
3//! This crate provides a rate limiter with integrated PID controller to manage request rates
4//! dynamically. It is designed for distributed systems where controlling the rate of requests
5//! is crucial for stability and performance.
6//!
7//! ## Example
8//!
9//! The following example demonstrates how to use the `RateLimiter` with the builder pattern to create
10//! a rate limiter instance and make throttling decisions based on the request rate.
11//!
12//! ```rust
13//! use nenya::RateLimiterBuilder;
14//! use nenya::pid_controller::PIDControllerBuilder;
15//! use std::time::Duration;
16//!
17//! // Create a PID controller with specific parameters
18//! let pid_controller = PIDControllerBuilder::new(10.0)
19//!     .kp(1.0)
20//!     .ki(0.1)
21//!     .kd(0.01)
22//!     .build();
23//!
24//! // Create a rate limiter using the builder pattern
25//! let mut rate_limiter = RateLimiterBuilder::new(10.0)
26//!     .min_rate(5.0)
27//!     .max_rate(15.0)
28//!     .pid_controller(pid_controller)
29//!     .update_interval(Duration::from_secs(1))
30//!     .build();
31//!
32//! // Simulate request processing and check if throttling is necessary
33//! for _ in 0..20 {
34//!     if rate_limiter.should_throttle() {
35//!         println!("Request throttled");
36//!     } else {
37//!         println!("Request accepted");
38//!     }
39//! }
40//! ```

74/// Adaptive-window floor for the sliding-window rate estimator: trimming
75/// keeps at least this many accepted timestamps, so below
76/// `min_window_samples / update_interval` the measurement window stretches
77/// to span the last N accepts instead of reading mostly-empty windows.
78///
79/// Simulator-derived (Milestone 5.4 sweep, seed 42, 300s runs, PID engine;
80/// re-run instructions in docs/capacity-model.md): steady over-admission at
81/// sparse per-node shares falls from +16–18% (K=0, the old fixed window)
82/// to +2.5–3.3% at K=20 across 0.75/3/5 rps/node shares, with healthy
83/// shares (≥100 rps/node) byte-identical. K=50 over-remembers at extreme
84/// sparsity (+5.4% at 0.75 rps/node); K=10 leaves ~5% bias. 20 is the
85/// swept optimum.
86const DEFAULT_MIN_WINDOW_SAMPLES: usize = 20;
87
88/// Floor on the adaptive bucket capacity, in tokens.
89///
90/// Two failure modes at sparse fair shares (a small per-user limit split
91/// across a large cluster) motivate it, both found in the Milestone 6
92/// large-cluster review:
93/// 1. **Starvation**: with `capacity = refill × 1s` alone, a share below
94///    1 rps yields a sub-token capacity — the bucket can never accumulate
95///    a whole token and admits nothing, ever.
96/// 2. **Poisson clump loss**: at exactly one token, an arrival is only
97///    admitted if a full `1/refill` gap preceded it, so a Poisson stream
98///    below the share still loses heavily to clumping.
99///
100/// Simulator-derived (tier_threshold_sweep axis 4, seed 42: an 8 rps user
101/// on 25 nodes against a 10 rps limit, offered *below* its fair share):
102/// served fraction 0.62 / 0.84 / 0.94 / 0.97 at floors 1 / 2 / 4 / 8
103/// tokens; service-scale scenarios (autoscale join overshoot, steady
104/// overage) are unchanged at every value because their capacities are
105/// hundreds of tokens. 4 is the knee. Cost: a cold bucket can burst up to
106/// `max(floor − refill × 1s, 0)` extra tokens per node.
107const CAPACITY_FLOOR_TOKENS: f64 = 4.0;
108
109/// Token bucket rate limiter with sliding window measurement and PID controller for adaptive coordination.
110///
111/// This hybrid architecture uses:
112/// - Token bucket for precise, collision-immune throttling decisions
113/// - Sliding window for accurate rate measurement (PID feedback)
114/// - PID controller for adaptive distributed coordination
115///
116/// # Coordination Modes
117///
118/// **Single-Node Mode** (default):
119/// - `cluster_target = None`: PID tracks local rate only
120/// - `num_peers = 0`: No distribution
121///
122/// **Distributed Mode** (multi-node cluster):
123/// - `cluster_target = Some(1000.0)`: Cluster-wide rate target
124/// - `num_peers` tracked via gossip: Number of peer nodes
125/// - Equal division PID: Each node takes proportional share of cluster error
126#[derive(Debug)]
127pub struct RateLimiter<T> {
128    // Token Bucket state (local throttling)
129    tokens: T,
130    last_refill: Instant,
131    refill_rate: T,
132    bucket_capacity: T,
133
134    // Sliding Window state (rate measurement)
135    accepted_request_timestamps: VecDeque<Instant>,
136    local_accepted_request_rate: T,
137
138    /// Adaptive-window floor: trimming keeps at least this many accepted
139    /// timestamps even when they fall outside `update_interval`, so the
140    /// rate estimate stays sample-based at sparse rates instead of reading
141    /// an empty window as zero (see the Milestone 5 estimator-floor fix)
142    min_window_samples: usize,
143
144    /// When set, `bucket_capacity` tracks the engine output after every
145    /// control update: `capacity = refill_rate × burst_seconds`. In
146    /// distributed mode this shrinks the burst allowance to the node's
147    /// current fair share instead of a full cluster target (the Milestone 5
148    /// cold-start fix); `None` keeps the static configured capacity.
149    bucket_burst_seconds: Option<T>,
150
151    // Control engine (adaptive coordination; PID by default)
152    engine: Box<dyn RateController<T>>,
153    target_rate: T,            // Base rate for this node (single-node mode)
154    cluster_target: Option<T>, // Cluster-wide target (distributed mode)
155    min_rate: T,
156    max_rate: T,
157    update_interval: Duration,
158    last_updated: Instant,
159
160    // Distributed coordination
161    external_accepted_request_rate: T,
162    num_peers: usize, // Number of peer nodes (0 = single-node mode)
163
164    /// Per-peer `(rate, age)` observations for this scope, consumed by the
165    /// control engine. When empty, the legacy `set_num_peers` /
166    /// `set_external_accepted_request_rate` values are synthesized into
167    /// observations so existing callers keep working.
168    peer_observations: Vec<PeerRate<T>>,
169}

196            local_accepted_request_rate: T::zero(),

200            // Control engine: the provided PID behind the trait
201            engine: Box::new(PidEngine::new(pid_controller)),
202            target_rate,
203            cluster_target: None, // Single-node by default
204            min_rate,
205            max_rate,
206            update_interval,
207            last_updated: now,
208
209            // Distributed coordination
210            external_accepted_request_rate: T::zero(),
211            num_peers: 0, // Single-node by default
212            peer_observations: Vec::new(),
213        }

216    /// Refills the token bucket based on elapsed time since last refill.
217    ///
218    /// Tokens are added based on: elapsed_time * refill_rate
219    /// The token count is capped at bucket_capacity to prevent unbounded burst.
220    fn refill_tokens(&mut self, now: Instant) {
221        let elapsed = now.duration_since(self.last_refill).as_secs_f64();
222
223        // Calculate new tokens based on refill rate
224        let new_tokens = T::from_f64(elapsed).unwrap() * self.refill_rate;
225        self.tokens = (self.tokens + new_tokens).min(self.bucket_capacity);
226
227        self.last_refill = now;
228    }
229
230    /// Determines if the current request should be throttled based on the rate limiter's state.
231    ///
232    /// This method uses the current system time (`Instant::now()`). For testing or scenarios
233    /// where you need to control time, use [`should_throttle_at`](Self::should_throttle_at).
234    ///
235    /// Returns `true` if the request should be throttled, `false` otherwise.
236    pub fn should_throttle(&mut self) -> bool {
237        self.should_throttle_at(Instant::now())
238    }
239
240    /// Determines if a request at the given timestamp should be throttled.
241    ///
242    /// This is the core rate limiting logic with injectable time. Uses a token bucket
243    /// for precise throttling decisions, immune to timestamp collisions.
244    ///
245    /// # Arguments
246    /// * `now` - The timestamp of the current request
247    ///
248    /// # Returns
249    /// `true` if the request should be throttled, `false` otherwise.
250    ///
251    /// # Example
252    /// ```rust
253    /// use nenya::RateLimiterBuilder;
254    /// use nenya::pid_controller::PIDControllerBuilder;
255    /// use std::time::{Duration, Instant};
256    ///
257    /// let pid = PIDControllerBuilder::new(100.0).kp(0.8).build();
258    /// let mut limiter = RateLimiterBuilder::new(100.0)
259    ///     .min_rate(50.0)
260    ///     .max_rate(150.0)
261    ///     .pid_controller(pid)
262    ///     .update_interval(Duration::from_millis(100))
263    ///     .build();
264    ///
265    /// let base_time = Instant::now();
266    /// // Simulate requests with controlled timestamps
267    /// for i in 0..10 {
268    ///     let request_time = base_time + Duration::from_millis(i * 10);
269    ///     let throttled = limiter.should_throttle_at(request_time);
270    ///     println!("Request {}: throttled={}", i, throttled);
271    /// }
272    /// ```
273    pub fn should_throttle_at(&mut self, now: Instant) -> bool {
274        // 1. Refill tokens based on elapsed time
275        self.refill_tokens(now);
276
277        // 2. Periodically update the control engine (regardless of accept/throttle decision)
278        if now.duration_since(self.last_updated) >= self.update_interval {
279            self.update_control(now);
280        }
281
282        // 3. Make throttle decision (token bucket)
283        if self.tokens >= T::one() {
284            self.tokens = self.tokens - T::one();
285
286            // Record acceptance for rate measurement
287            self.accepted_request_timestamps.push_back(now);
288            self.trim_accepted_timestamps(now);
289
290            false // Accept request
291        } else {
292            true // Throttle request
293        }
294    }
295
296    /// Updates internal state (trims windows, updates PID) at the given timestamp
297    /// without recording an actual request.
298    ///
299    /// This is useful for test scenarios where you want to keep the rate limiter's
300    /// state current during periods of zero load, without polluting the request windows
301    /// with fake timestamps.
302    ///
303    /// In production code, you typically only call `should_throttle()` or `should_throttle_at()`
304    /// when actual requests arrive.
305    pub fn update_state_at(&mut self, now: Instant) {
306        self.refill_tokens(now);
307        self.trim_accepted_timestamps(now);
308        self.calculate_local_accepted_rate(now);
309
310        // Update the control engine periodically
311        if now.duration_since(self.last_updated) >= self.update_interval {
312            self.update_control(now);
313        }
314    }
315
316    /// Runs one control-engine update: measures the local rate, hands the
317    /// engine the current observations, and applies the returned refill rate.
318    ///
319    /// The engine owns the meaningful clamping (in distributed mode the
320    /// bounds scale with the live node count, which only the engine knows);
321    /// this caller just sanitizes the output to a non-negative finite value.
322    ///
323    /// When no per-peer observations were provided (legacy callers using
324    /// `set_num_peers` + `set_external_accepted_request_rate`), equivalent
325    /// observations are synthesized: `num_peers` fresh peers splitting the
326    /// external rate evenly.
327    fn update_control(&mut self, now: Instant) {
328        // Measure local accepted rate (sliding window)
329        self.calculate_local_accepted_rate(now);
330
331        let synthesized: Vec<PeerRate<T>>;
332        let peers: &[PeerRate<T>] = if !self.peer_observations.is_empty() {
333            &self.peer_observations
334        } else if self.num_peers > 0 {
335            let per_peer =
336                self.external_accepted_request_rate / T::from_usize(self.num_peers).unwrap();
337            synthesized = (0..self.num_peers)
338                .map(|i| PeerRate {
339                    id: i.to_string(),
340                    rate: per_peer,
341                    age: Duration::ZERO,
342                })
343                .collect();
344            &synthesized
345        } else {
346            &[]
347        };
348
349        let input = ControlInput {
350            local_rate: self.local_accepted_request_rate,
351            peers,
352            target_rate: self.target_rate,
353            cluster_target: self.cluster_target,
354            min_rate: self.min_rate,
355            max_rate: self.max_rate,
356            dt: now.duration_since(self.last_updated),
357        };
358        let refill = self.engine.update(&input);
359        if refill.is_finite() {
360            self.refill_rate = refill.max(T::zero());
361        }
362
363        // Adaptive burst allowance: capacity follows the engine's rate, so
364        // a node's burst headroom is proportional to its current share.
365        // Floored at a few tokens so sparse shares (small per-user limit,
366        // large cluster) neither starve outright nor bleed admissions to
367        // Poisson clumping — see CAPACITY_FLOOR_TOKENS for the sweep.
368        if let Some(burst_seconds) = self.bucket_burst_seconds {
369            self.bucket_capacity =
370                (self.refill_rate * burst_seconds).max(T::from_f64(CAPACITY_FLOOR_TOKENS).unwrap());
371            self.tokens = self.tokens.min(self.bucket_capacity);
372        }
373
374        self.last_updated = now;
375    }
376
377    /// Calculates the local accepted request rate based on the sliding window.
378    ///
379    /// Uses a smaller minimum duration threshold (1ms) compared to the old implementation,
380    /// because token bucket throttling reduces timestamp collisions in the accepted stream.
381    fn calculate_local_accepted_rate(&mut self, now: Instant) {
382        if let Some(&oldest) = self.accepted_request_timestamps.front() {
383            let duration = now.duration_since(oldest).as_secs_f64();
384
385            // Smaller minimum duration acceptable (fewer collisions due to token bucket)
386            let effective_duration = duration.max(0.001); // 1ms minimum
387
388            self.local_accepted_request_rate =
389                T::from_usize(self.accepted_request_timestamps.len()).unwrap()
390                    / T::from_f64(effective_duration).unwrap();
391        } else {
392            self.local_accepted_request_rate = T::zero();
393        }
394    }
395
396    /// Trims old accepted request timestamps that are outside the update
397    /// interval, but always keeps the most recent `min_window_samples` so
398    /// the rate estimate stays sample-based at sparse rates (below
399    /// `min_window_samples / update_interval` the window stretches to span
400    /// the last N accepts; above it, behavior is unchanged).
401    fn trim_accepted_timestamps(&mut self, now: Instant) {
402        while self.accepted_request_timestamps.len() > self.min_window_samples {
403            match self.accepted_request_timestamps.front() {
404                Some(timestamp) if now.duration_since(*timestamp) > self.update_interval => {
405                    self.accepted_request_timestamps.pop_front();
406                }
407                _ => break,
408            }
409        }
410    }

561    /// Sets the *starting* bucket capacity without pinning it static:
562    /// unlike [`bucket_capacity`](Self::bucket_capacity), the adaptive
563    /// burst allowance stays enabled and the capacity converges to
564    /// `refill × burst_seconds` at the first control update. Used when a
565    /// tail scope promotes mid-burst: the tail bucket's depth carries
566    /// over so promotion doesn't truncate an in-flight burst.
567    pub fn initial_capacity(mut self, capacity: T) -> Self {
568        self.initial_capacity = Some(capacity);
569        self
570    }

572    /// Sets the initial token refill rate (default: `target_rate`).
573    ///
574    /// In distributed mode the refill rate only converges to the node's
575    /// fair share after the first control update, so a limiter created
576    /// mid-flight (e.g. a tail scope promoted into the hot tier) would
577    /// admit at the full cluster target for up to one `update_interval`.
578    /// Starting the refill at the already-known share removes that
579    /// discontinuity; the engine takes over from there.
580    pub fn initial_refill_rate(mut self, rate: T) -> Self {
581        self.initial_refill_rate = Some(rate);
582        self
583    }

585    /// Sets the initial token fill as a fraction of bucket capacity
586    /// (default 1.0 — start with a full bucket). `0.0` gives a cold-started
587    /// node no free burst: it admits only what its refill rate earns,
588    /// which in distributed mode prevents each joining node from admitting
589    /// a full cluster-target-sized burst before its first control update.
590    pub fn initial_tokens_frac(mut self, frac: T) -> Self {
591        self.initial_tokens_frac = Some(frac);
592        self
593    }

595    /// Makes the bucket capacity track the control engine's output:

598    /// staying at the configured cluster-scale capacity.
599    pub fn bucket_burst_seconds(mut self, seconds: T) -> Self {

602    }

612    }

721    ///

723    /// let throttled = limiter.should_throttle_at(base_time);

730    /// Builds and returns the `RateLimiter` instance.
731    pub fn build(self) -> RateLimiter<T> {
732        let now = self.initial_timestamp.unwrap_or_else(Instant::now);
733        let initial_refill = self.initial_refill_rate.unwrap_or(self.target_rate);
734        // Starting capacity precedence: explicit static capacity >
735        // explicit initial (adaptive) capacity > the initial refill (an
736        // explicit refill also scales the initial burst allowance —
737        // capacity tracks refill × 1s from the first control update
738        // anyway) > the target
739        let bucket_capacity = self.bucket_capacity.unwrap_or_else(|| {
740            self.initial_capacity
741                .unwrap_or(if self.initial_refill_rate.is_some() {
742                    initial_refill
743                } else {
744                    self.target_rate
745                })
746        });
747        let initial_tokens = match self.initial_tokens_frac {
748            Some(frac) => bucket_capacity * frac,
749            None => bucket_capacity,
750        };
751
752        // Burst allowance default: adaptive — capacity tracks the engine's
753        // refill rate (1 s of the *current* rate) unless the caller pinned
754        // an explicit static capacity. Simulator-derived (Milestone 5.4
755        // cold-start sweep, seed 42): with the static cluster-scale bucket,
756        // every cold-started node banks and admits a full cluster target of
757        // burst, which made fleet convergence linear in node count
758        // (37 s/72 s at 50/100 nodes → 4 s flat with tracking) and
759        // dominated join overshoot (autoscale 8040 → 2076 requests).
760        let bucket_burst_seconds = match (self.bucket_burst_seconds, self.bucket_capacity) {
761            (Some(seconds), _) => Some(seconds),
762            (None, Some(_)) => None, // explicit capacity stays static
763            (None, None) => Some(T::one()),
764        };
765
766        // For distributed mode with PID, setpoint will be adjusted dynamically per update
767        let initial_setpoint = if self.cluster_target.is_some() {
768            T::zero() // Will be set on the first engine update
769        } else {
770            self.target_rate
771        };
772
773        // Engine precedence: explicit engine > provided PID > static PID
774        let engine: Box<dyn RateController<T>> = match (self.engine, self.pid_controller) {
775            (Some(engine), _) => engine,
776            (None, Some(pid)) => Box::new(PidEngine::new(pid)),
777            (None, None) => Box::new(PidEngine::new(PIDController::new_static_controller(
778                initial_setpoint,
779            ))),
780        };
781
782        RateLimiter {
783            tokens: initial_tokens,
784            last_refill: now,
785            refill_rate: initial_refill,
786            bucket_capacity,
787
788            // Sliding window starts empty
789            accepted_request_timestamps: VecDeque::new(),
790            local_accepted_request_rate: T::zero(),
791            min_window_samples: self.min_window_samples,
792            bucket_burst_seconds,
793
794            // Control engine
795            engine,
796            target_rate: self.target_rate,
797            cluster_target: self.cluster_target,
798            min_rate: self.min_rate,
799            max_rate: self.max_rate,
800            update_interval: self.update_interval,
801            last_updated: now,
802
803            // Distributed coordination
804            external_accepted_request_rate: self.external_accepted_request_rate,
805            num_peers: self.num_peers,
806            peer_observations: Vec::new(),
807        }
808    }
