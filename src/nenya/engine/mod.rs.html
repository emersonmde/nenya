1//! Pluggable control engines (Milestone 5).
2//!
3//! A [`RateController`] turns observations — the local accepted rate plus
4//! per-peer `(rate, age)` samples — into a new local token refill rate. The
5//! trait boundary is deliberately narrow: engines see observations, not
6//! gossip internals, so the same engine runs unchanged under the Chitchat
7//! transport, the deterministic simulator, and any future blackboard
8//! transport. How an engine weighs staleness, divides the cluster target
9//! across nodes, and reacts to noise is *inside* the boundary — aggregation
10//! strategy is part of what engines compete on.
11//!
12//! Three engines ship:
13//!
14//! - [`PidEngine`] — the original equal-division PID loop, ported unchanged.
15//! - [`BayesianEngine`] — per-peer scalar Kalman filters; admits against an
16//!   upper confidence bound of the cluster estimate, so staleness maps to
17//!   uncertainty and the node is conservative exactly when information is
18//!   old.
19//! - [`HybridEngine`] — the Kalman peer estimate feeding a cluster-level PID.
20//!
21//! The engine is always an explicit config choice — never selected at
22//! runtime. See `docs/engine-comparison.md` for the benchmark data behind
23//! the recommended default.

36/// One peer's most recent accepted-rate observation for a single scope.
37///
38/// `age` is measured on the local monotonic clock (time since the
39/// observation was last seen to change) — peer wall-clock timestamps are
40/// never compared across nodes, so clock skew cannot affect engines.
41#[derive(Debug, Clone, PartialEq)]
42pub struct PeerRate<T> {
43    /// Stable peer identity. Engines that track per-peer state (Kalman
44    /// filters) key on this; it only needs to be unique and stable per peer.
45    pub id: String,
46
47    /// The peer's last reported accepted rate for this scope
48    pub rate: T,
49
50    /// Locally measured time since this observation last changed
51    pub age: Duration,
52}
53
54/// Everything an engine sees at one control update.
55///
56/// `min_rate`/`max_rate` are the configured cluster-scale bounds. Engines
57/// apply them (scaled by their own view of the live node count, which only
58/// the engine knows); the caller merely sanitizes the output to a
59/// non-negative finite value.
60#[derive(Debug)]
61pub struct ControlInput<'a, T> {
62    /// Locally measured accepted rate (sliding window)
63    pub local_rate: T,
64
65    /// Per-peer observations for this scope
66    pub peers: &'a [PeerRate<T>],
67
68    /// Local target rate (single-node mode)
69    pub target_rate: T,
70
71    /// Cluster-wide target (distributed mode); `None` = single-node
72    pub cluster_target: Option<T>,
73
74    /// Configured minimum rate bound (cluster scale)
75    pub min_rate: T,
76
77    /// Configured maximum rate bound (cluster scale)
78    pub max_rate: T,
79
80    /// Time elapsed since the previous engine update
81    pub dt: Duration,
82}

84/// A control engine: consumes one [`ControlInput`] per update interval and
85/// returns the new local token refill rate.
86///
87/// Updates run in the sync loop cadence (roughly once per second per
88/// scope), never on the per-request hot path.
89pub trait RateController<T>: fmt::Debug + Send + Sync {
90    /// Compute the new local refill rate from the current observations.
91    fn update(&mut self, input: &ControlInput<'_, T>) -> T;
92
93    /// The engine's current effective target for the local node (its share
94    /// of the cluster target in distributed mode).
95    fn setpoint(&self) -> T;
96}

98/// Which control engine to run.
99///
100/// Always an explicit configuration choice (`engine = "pid" | "bayesian" |
101/// "hybrid"`); there is no runtime auto-selection. Benchmarks only decide
102/// which value ships as the documented recommended default.
103#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
104#[cfg_attr(feature = "server", derive(serde::Serialize, serde::Deserialize))]
105#[cfg_attr(feature = "server", serde(rename_all = "lowercase"))]
106pub enum EngineKind {
107    #[default]
108    Pid,
109    Bayesian,
110    Hybrid,
111}
112
113impl std::str::FromStr for EngineKind {
114    type Err = String;
115
116    fn from_str(s: &str) -> Result<Self, Self::Err> {
117        match s.to_ascii_lowercase().as_str() {
118            "pid" => Ok(EngineKind::Pid),
119            "bayesian" => Ok(EngineKind::Bayesian),
120            "hybrid" => Ok(EngineKind::Hybrid),
121            other => Err(format!(
122                "unknown engine '{}' (expected pid, bayesian, or hybrid)",
123                other
124            )),
125        }
126    }
127}
128
129impl fmt::Display for EngineKind {
130    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
131        f.write_str(match self {
132            EngineKind::Pid => "pid",
133            EngineKind::Bayesian => "bayesian",
134            EngineKind::Hybrid => "hybrid",
135        })
136    }
137}

139/// Compute the staleness weight for an observation of the given age.
140///
141/// - Full weight (1.0) for ages up to `2 × sync_interval` — a healthy peer
142///   publishes every `sync_interval`, so anything fresher than two intervals
143///   is just normal propagation delay, not staleness
144/// - Linear decay from 1.0 to 0.0 between `2 × sync_interval` and
145///   `stale_timeout`
146/// - Zero at and beyond `stale_timeout` — the peer is presumed dead or
147///   partitioned and its last known rate is phantom load
148///
149/// Degenerate configurations (`stale_timeout <= 2 × sync_interval`) collapse
150/// to a hard cutoff at `stale_timeout`.
151///
152/// Lives in the engine module (always compiled) because both the gossip
153/// aggregation layer and the [`PidEngine`] liveness test use it.
154pub fn staleness_weight(age: Duration, sync_interval: Duration, stale_timeout: Duration) -> f64 {
155    if age >= stale_timeout {
156        return 0.0;
157    }
158
159    let full_weight_window = 2 * sync_interval;
160    if age <= full_weight_window {
161        return 1.0;
162    }
163
164    let decay_span = stale_timeout.saturating_sub(full_weight_window);
165    if decay_span.is_zero() {
166        // Degenerate config: no decay span, hard cutoff at stale_timeout
167        return 1.0;
168    }
169
170    let into_decay = age - full_weight_window;
171    1.0 - into_decay.as_secs_f64() / decay_span.as_secs_f64()
172}
