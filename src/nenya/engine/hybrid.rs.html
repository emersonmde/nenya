1//! Hybrid engine: Kalman-filtered cluster estimate feeding a cluster-level
2//! PID loop.
3//!
4//! The estimator half is the same per-peer filter bank as
5//! [`BayesianEngine`](super::BayesianEngine) (see that module for the model
6//! and citations). The control half differs from [`PidEngine`](super::PidEngine)
7//! in two ways:
8//!
9//! 1. **The feedback signal is the estimated cluster total**
10//!    (`local_rate + Σ peer means`), not the local rate alone. The filter
11//!    smooths gossip sampling noise before it reaches the controller.
12//! 2. **Control runs at cluster level, divided by n**: every node computes
13//!    the same cluster error `cluster_target − estimate`, and applies
14//!    `correction / n` around its `target / n` share. Summed over n nodes
15//!    this reproduces exactly one controller's worth of aggregate response,
16//!    so the loop gain — and settling time — is independent of fleet size,
17//!    unlike equal-division PID where each node integrates only its `1/n`
18//!    error slice. The same configured gains remain valid because the
19//!    per-node output magnitude is unchanged.
20//!
21//! No optimality is claimed: the separation principle that would make
22//! estimate-then-control provably optimal assumes a linear plant, Gaussian
23//! noise, and no delay, all of which gossip coordination violates. The
24//! engine earns its place (or not) in the Milestone 5 scenario matrix.
25

32/// Estimator parameters for [`HybridEngine`]. The admission-side
33/// `confidence_z` is unused here (the PID consumes the mean estimate);
34/// process/measurement noise and the membership horizon apply as in
35/// [`BayesianParams`].
36pub type HybridParams = BayesianParams;
37
38/// Kalman→PID hybrid. See the module docs.
39#[derive(Debug, Clone)]
40pub struct HybridEngine<T> {
41    bank: FilterBank<T>,
42    pid: PIDController<T>,
43    last_setpoint: Option<T>,
44}
45
46impl<T: Float + FromPrimitive + Copy> HybridEngine<T> {
47    pub fn new(pid: PIDController<T>, params: HybridParams) -> Self {
48        HybridEngine {
49            bank: FilterBank::new(&params),
50            pid,
51            last_setpoint: None,
52        }
53    }
54}
55
56impl<T: Float + Signed + FromPrimitive + Copy + Send + Sync + std::fmt::Debug> RateController<T>
57    for HybridEngine<T>
58{
59    fn update(&mut self, input: &ControlInput<'_, T>) -> T {
60        let Some(cluster_target) = input.cluster_target else {
61            // Single-node mode: no peers to estimate; plain local PID
62            self.pid.set_setpoint(input.target_rate);
63            let correction = self.pid.compute_correction(input.local_rate);
64            let refill = num_traits::clamp(
65                input.target_rate + correction,
66                input.min_rate,
67                input.max_rate,
68            );
69            self.last_setpoint = Some(input.target_rate);
70            return refill;
71        };
72
73        let estimates = self.bank.observe(input.peers, input.dt);
74        let live = estimates.len();
75        let n = T::from_usize(1 + live).unwrap();
76
77        let mut peer_sum = T::zero();
78        for e in &estimates {
79            peer_sum = peer_sum + e.mean;
80        }
81        let cluster_estimate = input.local_rate + peer_sum;
82
83        // Cluster-level PID, per-node share of the correction
84        self.pid.set_setpoint(cluster_target);
85        let correction = self.pid.compute_correction(cluster_estimate);
86
87        let share = cluster_target / n;
88        self.last_setpoint = Some(share);
89        num_traits::clamp(
90            share + correction / n,
91            input.min_rate / n,
92            input.max_rate / n,
93        )
94    }
95
96    fn setpoint(&self) -> T {
97        self.last_setpoint.unwrap_or_else(T::zero)
98    }
99}

101#[cfg(test)]
102mod tests {
103    use super::*;
104    use crate::engine::PeerRate;

108    fn peer(id: &str, rate: f64, age_ms: u64) -> PeerRate<f64> {
109        PeerRate {
110            id: id.to_string(),
111            rate,
112            age: Duration::from_millis(age_ms),
113        }
114    }
115
116    fn input<'a>(
117        local_rate: f64,
118        peers: &'a [PeerRate<f64>],
119        cluster_target: Option<f64>,
120    ) -> ControlInput<'a, f64> {
121        ControlInput {
122            local_rate,
123            peers,
124            target_rate: 100.0,
125            cluster_target,
126            min_rate: 30.0,
127            max_rate: 600.0,
128            dt: Duration::from_secs(1),
129        }
130    }
131
132    #[test]
133    fn test_at_target_no_correction() {
134        let pid = PIDControllerBuilder::new(0.0).kp(0.5).build();
135        let mut engine = HybridEngine::new(pid, HybridParams::default());
136        // Cluster estimate = 100 local + 200 peers = 300 = target
137        let peers = vec![peer("a", 100.0, 0), peer("b", 100.0, 0)];
138        let refill = engine.update(&input(100.0, &peers, Some(300.0)));
139        assert_eq!(refill, 100.0, "zero error → share");
140        assert_eq!(engine.setpoint(), 100.0);
141    }
142
143    #[test]
144    fn test_over_target_reduces_share() {
145        let pid = PIDControllerBuilder::new(0.0).kp(0.5).build();
146        let mut engine = HybridEngine::new(pid, HybridParams::default());
147        // Estimate 450 vs target 300: correction = 0.5·(−150) = −75, /3 = −25
148        let peers = vec![peer("a", 150.0, 0), peer("b", 150.0, 0)];
149        let refill = engine.update(&input(150.0, &peers, Some(300.0)));
150        assert_eq!(refill, 75.0);
151    }
152
153    #[test]
154    fn test_correction_split_is_n_independent_in_aggregate() {
155        // n nodes each applying correction/n must sum to one controller's
156        // output regardless of n
157        for n in [2usize, 5, 10] {
158            let pid = PIDControllerBuilder::new(0.0).kp(1.0).build();
159            let mut engine = HybridEngine::new(pid, HybridParams::default());
160            let peers: Vec<_> = (0..n - 1)
161                .map(|i| peer(&format!("p{}", i), 200.0 / n as f64, 0))
162                .collect();
163            let local = 200.0 / n as f64;
164            let refill = engine.update(&input(local, &peers, Some(300.0)));
165            let share = 300.0 / n as f64;
166            let aggregate_correction = (refill - share) * n as f64;
167            // Estimate = 200 total, error = +100, kp = 1 → aggregate +100
168            assert!(
169                (aggregate_correction - 100.0).abs() < 1e-6,
170                "n={}: aggregate correction {}",
171                n,
172                aggregate_correction
173            );
174        }
175    }
176
177    #[test]
178    fn test_single_node_mode_matches_pid_semantics() {
179        let pid = PIDControllerBuilder::new(100.0).kp(0.5).build();
180        let mut engine = HybridEngine::new(pid, HybridParams::default());
181        let refill = engine.update(&input(80.0, &[], None));
182        // error 20 → correction 10 → 110
183        assert_eq!(refill, 110.0);
184    }
